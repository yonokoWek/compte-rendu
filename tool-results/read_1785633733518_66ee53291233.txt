    85→    85→
    86→    86→export default function CompteRenduTab() {
    87→    87→  const period = useAppStore((s) => s.period);
    88→    88→  const periodType = useAppStore((s) => s.periodType);
    89→    89→  const setPeriodType = useAppStore((s) => s.setPeriodType);
    90→    90→  const nextPeriod = useAppStore((s) => s.nextPeriod);
    91→    91→  const prevPeriod = useAppStore((s) => s.prevPeriod);
    92→    92→  const goToCurrentPeriod = useAppStore((s) => s.goToCurrentPeriod);
    93→    93→  const setProfileDialogOpen = useAppStore((s) => s.setProfileDialogOpen);
    94→    94→  const queryClient = useQueryClient();
    95→    95→
    96→    96→  // State for management dialog
    97→    97→  const [manageOpen, setManageOpen] = useState(false);
    98→    98→  const [newGroupName, setNewGroupName] = useState('');
    99→    99→  const [newCatName, setNewCatName] = useState('');
   100→   100→  const [newCatUnit, setNewCatUnit] = useState('minutes');
   101→   101→  const [newCatPersonal, setNewCatPersonal] = useState(true);
   102→   102→  const [assigningCat, setAssigningCat] = useState<Category | null>(null);
   103→   103→
   104→   104→  // Fetch categories with groups
   105→   105→  const { data: catData } = useQuery<{
   106→   106→    categories: Category[];
   107→   107→    groups: Group[];
   108→   108→  }>({
   109→   109→    queryKey: ['categories'],
   110→   110→    queryFn: () => authFetch('/api/categories').then((r) => r.json()),
   111→   111→  });
   112→   112→  const categories = catData?.categories || [];
   113→   113→  const groups = catData?.groups || [];
   114→   114→
   115→   115→  // Fetch entries
   116→   116→  const { data: entries = [], isLoading: entriesLoading } = useQuery<DailyEntry[]>({
   117→   117→    queryKey: ['entries', period.startDate, period.endDate],
   118→   118→    queryFn: () =>
   119→   119→      authFetch(
   120→   120→        `/api/entries?startDate=${format(period.startDate, 'yyyy-MM-dd')}&endDate=${format(period.endDate, 'yyyy-MM-dd')}`
   121→   121→      ).then((r) => r.json()),
   122→   122→  });
   123→   123→
   124→   124→  // Fetch profile
   125→   125→  const { data: profile } = useQuery<UserProfile>({
   126→   126→    queryKey: ['profile'],
   127→   127→    queryFn: () => authFetch('/api/profile').then((r) => r.json()),
   128→   128→  });
   129→   129→
   130→   130→  // Save entry mutation
   131→   131→  const saveEntry = useMutation({
   132→   132→    mutationFn: (data: { date: string; categoryId: string; value: number }) =>
   133→   133→      authFetch('/api/entries', {
   134→   134→        method: 'POST',
   135→   135→        body: JSON.stringify(data),
   136→   136→      }).then((r) => r.json()),
   137→   137→    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['entries'] }),
   138→   138→  });
   139→   139→
   140→   140→  // Category mutations
   141→   141→  const addCategory = useMutation({
   142→   142→    mutationFn: (data: { type: 'category'; name: string; unit: string; isPersonal: boolean; groupId?: string }) =>
   143→   143→      authFetch('/api/categories', { method: 'POST', body: JSON.stringify(data) }).then((r) => r.json()),
   144→   144→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: ['categories'] }); setNewCatName(''); },
   145→   145→  });
   146→   146→
   147→   147→  const deleteCategory = useMutation({
   148→   148→    mutationFn: (id: string) => authFetch(`/api/categories?type=category&id=${id}`, { method: 'DELETE' }),
   149→   149→    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['categories'] }),
   150→   150→  });
   151→   151→
   152→   152→  const addGroup = useMutation({
   153→   153→    mutationFn: (name: string) =>
   154→   154→      authFetch('/api/categories', { method: 'POST', body: JSON.stringify({ type: 'group', name }) }).then((r) => r.json()),
   155→   155→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: ['categories'] }); setNewGroupName(''); toast.success('Groupe créé'); },
   156→   156→  });
   157→   157→
   158→   158→  const deleteGroup = useMutation({
   159→   159→    mutationFn: (id: string) => authFetch(`/api/categories?type=group&id=${id}`, { method: 'DELETE' }),
   160→   160→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: ['categories'] }); toast.success('Groupe supprimé'); },
   161→   161→  });
   162→   162→
   163→   163→  const assignToGroup = useMutation({
   164→   164→    mutationFn: (data: { type: 'assign'; categoryId: string; groupId: string | null }) =>
   165→   165→      authFetch('/api/categories', { method: 'POST', body: JSON.stringify(data) }).then((r) => r.json()),
   166→   166→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: ['categories'] }); setAssigningCat(null); toast.success('Activité déplacée'); },
   167→   167→  });
   168→   168→
   169→   169→  // Calculate columns
   170→   170→  const totalDays = differenceInCalendarDays(period.endDate, period.startDate) + 1;
   171→   171→  const isSingleWeek = totalDays <= 7;
   172→   172→
   173→   173→  let columns: { key: string; label: string; date: string }[] = [];
   174→   174→  if (isSingleWeek) {
   175→   175→    columns = getDaysInRange(period.startDate, period.endDate).map((d) => {
   176→   176→      const dayIndex = getDay(d);
   177→   177→      return { key: format(d, 'yyyy-MM-dd'), label: DAY_NAMES_SHORT[dayIndex === 0 ? 6 : dayIndex - 1], date: format(d, 'yyyy-MM-dd') };
   178→   178→    });
   179→   179→  } else {
   180→   180→    columns = getWeeksInRange(period.startDate, period.endDate).map((w, i) => ({
   181→   181→      key: `week-${i}`,
   182→   182→      label: `Sem ${i + 1}`,
   183→   183→      date: `${format(w.start, 'yyyy-MM-dd')}_${format(w.end, 'yyyy-MM-dd')}`,
   184→   184→    }));
   185→   185→  }
   186→   186→
   187→   187→  const entryMap: Record<string, number> = {};
   188→   188→  for (const e of entries) entryMap[`${e.date}_${e.categoryId}`] = e.value;
   189→   189→
   190→   190→  const getCellValue = (categoryId: string, col: typeof columns[0]) => {
   191→   191→    if (isSingleWeek) return entryMap[`${col.date}_${categoryId}`] || 0;
   192→   192→    const [wStart, wEnd] = col.date.split('_');
   193→   193→    let total = 0;
   194→   194→    for (const day of eachDayOfInterval({ start: new Date(wStart), end: new Date(wEnd) })) {
   195→   195→      total += entryMap[`${format(day, 'yyyy-MM-dd')}_${categoryId}`] || 0;
   196→   196→    }
   197→   197→    return total;
   198→   198→  };
   199→   199→
   200→   200→  const getRowTotal = (catId: string) => columns.reduce((s, col) => s + getCellValue(catId, col), 0);
   201→   201→
   202→   202→  const personalCategories = categories.filter((c) => c.isPersonal && c.unit === 'minutes');
   203→   203→  const getPersonalTotal = (col?: typeof columns[0]) => {
   204→   204→    if (col) return personalCategories.reduce((s, cat) => s + getCellValue(cat.id, col), 0);
   205→   205→    return columns.reduce((s, col) => s + personalCategories.reduce((s2, cat) => s2 + getCellValue(cat.id, col), 0), 0);
   206→   206→  };
   207→   207→
   208→   208→  const handleCellChange = (categoryId: string, columnKey: string, value: string) => {
   209→   209→    const numValue = parseInt(value) || 0;
   210→   210→    if (isSingleWeek) {
   211→   211→      saveEntry.mutate({ date: columnKey, categoryId, value: numValue });
   212→   212→    } else {
   213→   213→      const [wStart] = columnKey.split('_');
   214→   214→      saveEntry.mutate({ date: wStart, categoryId, value: numValue });
   215→   215→    }
   216→   216→  };
   217→   217→
   218→   218→  // PDF generation
   219→   219→  const [pdfLoading, setPdfLoading] = useState(false);
   220→   220→  const handleExportPDF = async () => {
   221→   221→    setPdfLoading(true);
   222→   222→    try {
   223→   223→      const reportRes = await authFetch('/api/report', {
   224→   224→        method: 'POST',
   225→   225→        body: JSON.stringify({ startDate: format(period.startDate, 'yyyy-MM-dd'), endDate: format(period.endDate, 'yyyy-MM-dd') }),
   226→   226→      });
   227→   227→      const reportData = await reportRes.json();
   228→   228→      const pdfRes = await authFetch('/api/generate-pdf', {
   229→   229→        method: 'POST',
   230→   230→        body: JSON.stringify({ html: reportData.html }),
   231→   231→      });
   232→   232→      const pdfData = await pdfRes.json();
   233→   233→      const link = document.createElement('a');
   234→   234→      link.href = pdfData.pdf;
   235→   235→      link.download = `compte-rendu-${format(period.startDate, 'yyyy-MM-dd')}.pdf`;
   236→   236→      link.click();
   237→   237→      toast.success('PDF exporté');
   238→   238→    } catch {
   239→   239→      toast.error('Erreur PDF');
   240→   240→    } finally {
   241→   241→      setPdfLoading(false);
   242→   242→    }
   243→   243→  };
   244→   244→
   245→   245→  const displayName = profile ? `${profile.firstName} ${profile.lastName}`.trim() : 'Mon Compte';
   246→   246→
   247→   247→  // Group categories for display
   248→   248→  const groupedDisplay = new Map<string, Category[]>();
   249→   249→  const ungrouped: Category[] = [];
   250→   250→  for (const cat of categories) {
   251→   251→    if (cat.groupId && cat.group) {
   252→   252→      if (!groupedDisplay.has(cat.groupId)) groupedDisplay.set(cat.groupId, []);
   253→   253→      groupedDisplay.get(cat.groupId)!.push(cat);
   254→   254→    } else {
   255→   255→      ungrouped.push(cat);
   256→   256→    }
   257→   257→  }
   258→   258→
   259→   259→  return (
   260→   260→    <div className="max-w-6xl mx-auto p-4 space-y-4">
   261→   261→      {/* Header */}
   262→   262→      <div className="flex items-center justify-between">
   263→   263→        <div className="flex items-center gap-2">
   264→   264→          <div className="bg-[var(--theme-primary-light)] p-2 rounded-full">
   265→   265→            <CalendarDays className="h-5 w-5 text-[var(--theme-primary)]" />
   266→   266→          </div>
   267→   267→          <div>
   268→   268→            <h1 className="text-lg font-bold text-gray-900">Compte Rendu</h1>
   269→   269→            <p className="text-xs text-gray-500">{displayName} {profile?.assembly ? `• ${profile.assembly}` : ''}</p>
   270→   270→          </div>
   271→   271→        </div>
   272→   272→        <div className="flex items-center gap-1">
   273→   273→          <Button variant="ghost" size="icon" onClick={() => setManageOpen(true)}>
   274→   274→            <Settings className="h-4 w-4" />
   275→   275→          </Button>
   276→   276→          <Button variant="ghost" size="icon" onClick={() => setProfileDialogOpen(true)}>
   277→   277→            <User className="h-4 w-4" />
   278→   278→          </Button>
   279→   279→          <Button variant="outline" size="sm" onClick={handleExportPDF} disabled={pdfLoading} className="gap-1 text-[var(--theme-primary)] border-[var(--theme-primary)] hover:bg-[var(--theme-primary-light)]">
   280→   280→            <Download className="h-4 w-4" />
   281→   281→            <span className="hidden sm:inline">PDF</span>
   282→   282→          </Button>
   283→   283→        </div>
   284→   284→      </div>
   285→   285→
   286→   286→      {/* Period selector */}
   287→   287→      <Card className="border-[var(--theme-primary)]">
   288→   288→        <CardContent className="p-3">
   289→   289→          <div className="flex items-center justify-between gap-2">
   290→   290→            <Button variant="ghost" size="icon" className="shrink-0" onClick={prevPeriod}>
   291→   291→              <ChevronLeft className="h-4 w-4" />
   292→   292→            </Button>
   293→   293→            <div className="flex-1 flex items-center justify-center gap-2">
   294→   294→              <Select value={periodType} onValueChange={(v) => setPeriodType(v as 'week' | 'month' | 'year')}>
   295→   295→                <SelectTrigger className="w-28 h-8 text-xs"><SelectValue /></SelectTrigger>
   296→   296→                <SelectContent>
   297→   297→                  <SelectItem value="week">Semaine</SelectItem>
   298→   298→                  <SelectItem value="month">Mois</SelectItem>
   299→   299→                  <SelectItem value="year">Année</SelectItem>
   300→   300→                </SelectContent>
   301→   301→              </Select>
   302→   302→              <span className="text-sm font-medium text-gray-800 truncate">{period.label}</span>
   303→   303→            </div>
   304→   304→            <div className="flex items-center gap-1">
   305→   305→              <Button variant="ghost" size="sm" className="text-xs h-8 px-2" onClick={goToCurrentPeriod}>Auj.</Button>
   306→   306→              <Button variant="ghost" size="icon" className="shrink-0" onClick={nextPeriod}>
   307→   307→                <ChevronRight className="h-4 w-4" />
   308→   308→              </Button>
   309→   309→            </div>
   310→   310→          </div>
   311→   311→        </CardContent>
   312→   312→      </Card>
   313→   313→
   314→   314→      {/* Main data table */}

   320→   320→                <tr>
   321→   321→                  <th className="sticky left-0 bg-[var(--theme-primary)] text-white px-2 py-2 text-left font-semibold text-xs min-w-[130px] z-10 rounded-tl-lg">Activité</th>
   322→   322→                  <th className="bg-[var(--theme-primary)] text-white px-1 py-2 text-center font-semibold text-xs w-10">{isSingleWeek ? 'Min' : ''}</th>
   323→   323→                  {columns.map((col) => (

   325→   325→                  ))}

   328→   328→              </thead>
   329→   329→              <tbody>
   330→   330→                {entriesLoading ? (
   331→   331→                  Array.from({ length: 8 }).map((_, i) => (
   332→   332→                    <tr key={i}>
   333→   333→                      <td className="bg-[var(--theme-primary-light)] px-2 py-2"><Skeleton className="h-4 w-24" /></td>
   334→   334→                      {columns.map((_, j) => <td key={j} className="px-2 py-2"><Skeleton className="h-6 w-8 mx-auto" /></td>)}
   335→   335→                    </tr>

   349→   349→                          </tr>
   350→   350→                          {cats.map((cat, rowIndex) => (

   352→   352→                              <td className="sticky left-0 px-2 py-1.5 text-left font-medium text-xs bg-inherit z-10 whitespace-nowrap border-r border-[var(--theme-primary-light)]">
   353→   353→                                <span className="text-[10px] text-[var(--theme-primary)]">●</span> {cat.name}

   364→   364→                                      className="w-12 h-7 text-center text-xs bg-transparent border border-transparent rounded hover:border-[var(--theme-primary)] focus:border-[var(--theme-primary)] focus:bg-[var(--theme-primary-light)] focus:outline-none transition-colors" />
   365→   365→                                  </td>
   366→   366→                                );
   367→   367→                              })}
   368→   368→                              <td className="px-2 py-1.5 text-center font-bold text-xs bg-[var(--theme-primary-light)]/80">

   370→   370→                              </td>
   371→   371→                            </tr>
   372→   372→                          ))}
   373→   373→                        </React.Fragment>
   374→   374→                      );
   375→   375→                    })}

   377→   377→                    {/* Ungrouped rows */}
   378→   378→                    {ungrouped.length > 0 && (
   379→   379→                      <>
   380→   380→                        <tr>
   381→   381→                          <td colSpan={columns.length + 3} className="bg-gray-200 text-gray-600 px-3 py-1 text-left text-xs font-semibold">
   382→   382→                            Sans groupe
   383→   383→                          </td>
   384→   384→                        </tr>
   385→   385→                        {ungrouped.map((cat, rowIndex) => (
   386→   386→                          <tr key={cat.id} className={rowIndex % 2 === 0 ? 'bg-white' : 'bg-[var(--theme-primary-light)]/50'}>
   387→   387→                            <td className="sticky left-0 px-2 py-1.5 text-left font-medium text-xs bg-inherit z-10 whitespace-nowrap border-r border-[var(--theme-primary-light)]">
   388→   388→                              <span className="text-[10px] text-gray-400">●</span> {cat.name}
   389→   389→                            </td>
   390→   390→                            <td className="px-1 py-1.5 text-center text-[10px] text-gray-400 bg-[var(--theme-primary-light)]/50">
   391→   391→                              {cat.unit === 'minutes' ? 'min' : 'Part.'}
   392→   392→                            </td>
   393→   393→                            {columns.map((col) => {

   395→   395→                              return (
   396→   396→                                <td key={col.key} className="px-1 py-1 text-center">

   398→   398→                                    placeholder="0"
   399→   399→                                    className="w-12 h-7 text-center text-xs bg-transparent border border-transparent rounded hover:border-[var(--theme-primary)] focus:border-[var(--theme-primary)] focus:bg-[var(--theme-primary-light)] focus:outline-none transition-colors" />
   400→   400→                                </td>
