   121→// Floating Note Component with proper drag handling
   122→function FloatingNote({ note, onUpdate, onDelete }: {
   123→  note: ReadingNote;
   124→  onUpdate: (id: string, content: string, positionX: number, positionY: number) => void;
   125→  onDelete: (id: string) => void;
   126→}) {
   127→  const noteRef = useRef<HTMLDivElement>(null);
   128→  const [editing, setEditing] = useState(false);
   129→  const isDragging = useRef(false);
   130→  const startPos = useRef({ x: 0, y: 0 });
   131→  const startMousePos = useRef({ x: 0, y: 0 });
   132→  const [editText, setEditText] = useState(note.content);
   133→
   134→  const startEditing = useCallback(() => {
   135→    setEditText(note.content);
   136→    setEditing(true);
   137→  }, [note.content]);
   138→
   139→  const handleMouseDown = useCallback((e: React.MouseEvent) => {
   140→    if (editing) return; // Don't drag while editing
   141→    e.preventDefault();
   142→    isDragging.current = false;
   143→    startPos.current = { x: note.positionX, y: note.positionY };
   144→    startMousePos.current = { x: e.clientX, y: e.clientY };
   145→
   146→    const handleMove = (me: MouseEvent) => {
   147→      const dx = me.clientX - startMousePos.current.x;
   148→      const dy = me.clientY - startMousePos.current.y;
   149→      if (Math.abs(dx) > 3 || Math.abs(dy) > 3) {
   150→        isDragging.current = true;
   151→      }
   152→      if (!isDragging.current || !noteRef.current) return;
   153→      const parent = noteRef.current.parentElement;
   154→      if (!parent) return;
   155→      const rect = parent.getBoundingClientRect();
   156→      const newX = Math.max(0, Math.min(95, ((startPos.current.x / 100) * rect.width + dx) / rect.width * 100));
   157→      const newY = Math.max(0, Math.min(90, ((startPos.current.y / 100) * rect.height + dy) / rect.height * 100));
   158→      noteRef.current.style.left = `${newX}%`;
   159→      noteRef.current.style.top = `${newY}%`;
   160→    };
   161→
   162→    const handleUp = (me: MouseEvent) => {
   163→      document.removeEventListener('mousemove', handleMove);
   164→      document.removeEventListener('mouseup', handleUp);
   165→      if (isDragging.current && noteRef.current) {
   166→        const left = parseFloat(noteRef.current.style.left) || note.positionX;
   167→        const top = parseFloat(noteRef.current.style.top) || note.positionY;
   168→        onUpdate(note.id, note.content, left, top);
   169→      }
   170→      isDragging.current = false;
   171→    };
   172→
   173→    document.addEventListener('mousemove', handleMove);
   174→    document.addEventListener('mouseup', handleUp);
   175→  }, [note.id, note.content, note.positionX, note.positionY, editing, onUpdate]);
   176→
   177→  // Touch support for mobile
   178→  const handleTouchStart = useCallback((e: React.TouchEvent) => {
   179→    if (editing) return;
   180→    const touch = e.touches[0];
   181→    isDragging.current = false;
   182→    startPos.current = { x: note.positionX, y: note.positionY };
   183→    startMousePos.current = { x: touch.clientX, y: touch.clientY };
   184→
   185→    const handleTouchMove = (te: TouchEvent) => {
   186→      const t = te.touches[0];
   187→      const dx = t.clientX - startMousePos.current.x;
   188→      const dy = t.clientY - startMousePos.current.y;
   189→      if (Math.abs(dx) > 3 || Math.abs(dy) > 3) {
   190→        isDragging.current = true;
   191→      }
   192→      if (!isDragging.current || !noteRef.current) return;
   193→      const parent = noteRef.current.parentElement;
   194→      if (!parent) return;
   195→      const rect = parent.getBoundingClientRect();
   196→      const newX = Math.max(0, Math.min(95, ((startPos.current.x / 100) * rect.width + dx) / rect.width * 100));
   197→      const newY = Math.max(0, Math.min(90, ((startPos.current.y / 100) * rect.height + dy) / rect.height * 100));
   198→      noteRef.current.style.left = `${newX}%`;
   199→      noteRef.current.style.top = `${newY}%`;
   200→    };
   201→
   202→    const handleTouchEnd = () => {
   203→      document.removeEventListener('touchmove', handleTouchMove);
   204→      document.removeEventListener('touchend', handleTouchEnd);
   205→      if (isDragging.current && noteRef.current) {
   206→        const left = parseFloat(noteRef.current.style.left) || note.positionX;
   207→        const top = parseFloat(noteRef.current.style.top) || note.positionY;
   208→        onUpdate(note.id, note.content, left, top);
   209→      }
   210→      isDragging.current = false;
   211→    };
   212→
   213→    document.addEventListener('touchmove', handleTouchMove, { passive: true });
   214→    document.addEventListener('touchend', handleTouchEnd);
   215→  }, [note.id, note.content, note.positionX, note.positionY, editing, onUpdate]);
   216→
   217→  const handleSave = () => {
   218→    onUpdate(note.id, editText, note.positionX, note.positionY);
   219→    setEditing(false);
   220→  };
   221→
   222→  return (
   223→    <div
   224→      ref={noteRef}
   225→      className="absolute z-50 select-none"
   226→      style={{ left: `${note.positionX}%`, top: `${note.positionY}%` }}
   227→    >
   228→      <div className="bg-yellow-100 border-2 border-yellow-300 rounded-lg shadow-lg p-2 min-w-[130px] max-w-[180px] transition-shadow hover:shadow-xl">
   229→        <div className="flex items-center justify-between mb-1">
   230→          <GripVertical className="h-3 w-3 text-yellow-400 cursor-grab active:cursor-grabbing" />
   231→          <div className="flex items-center gap-1">
   232→            <button
   233→              onClick={(e) => { e.stopPropagation(); if (editing) { handleSave(); } else { startEditing(); } }}
   234→              className="text-yellow-400 hover:text-[var(--theme-primary)]"
   235→            >
   236→              <PenLine className="h-2.5 w-2.5" />
   237→            </button>
   238→            <button
   239→              onClick={(e) => { e.stopPropagation(); onDelete(note.id); }}
   240→              className="text-yellow-400 hover:text-red-500"
   241→            >
   242→              <X className="h-2.5 w-2.5" />
   243→            </button>
   244→          </div>
   245→        </div>
   246→        {note.bibleRef && (
   247→          <p className="text-[8px] text-[var(--theme-primary)] font-bold mb-0.5">📖 {note.bibleRef}</p>
   248→        )}
   249→        {editing ? (
   250→          <textarea
   251→            value={editText}
   252→            onChange={(e) => setEditText(e.target.value)}
   253→            onBlur={handleSave}
   254→            onKeyDown={(e) => { if (e.key === 'Escape') setEditing(false); if (e.key === 'Enter' && e.metaKey) handleSave(); }}
   255→            className="w-full text-[10px] bg-transparent border border-yellow-200 rounded p-1 outline-none resize-none h-16 focus:border-[var(--theme-primary)]"
   256→            autoFocus
   257→            onClick={(e) => e.stopPropagation()}
   258→            onMouseDown={(e) => e.stopPropagation()}
   259→          />
   260→        ) : (
   261→          <p
   262→            onClick={(e) => { e.stopPropagation(); startEditing(); }}
   263→            className="text-[10px] text-gray-700 cursor-text leading-tight"
   264→          >
   265→            {note.content}
   266→          </p>
   267→        )}
   268→      </div>
   269→    </div>
   270→  );
   271→}

   810→              if (v === 'general') setNoteContext({});
   811→              else if (v === 'bible') setNoteContext({ bibleRef: 'Bible' });
   812→              else if (v === 'book' && books.length > 0) setNoteContext({ bookId: books[0].id });
   813→            }}>
   814→              <SelectTrigger className="h-9 text-xs"><SelectValue placeholder="Type de note" /></SelectTrigger>
   815→              <SelectContent>
   816→                <SelectItem value="general">📝 Note générale</SelectItem>
   817→                <SelectItem value="bible">📖 Note Bible</SelectItem>
   818→                {books.length > 0 && <SelectItem value="book">📚 Note livre</SelectItem>}
   819→              </SelectContent>
   820→            </Select>
   821→          )}
   822→          {(noteContext.bibleRef || noteContext.bibleRef === '') && noteContext.bookId === undefined && (
   823→            <Select value={noteContext.bibleRef || ''} onValueChange={(v) => setNoteContext({ bibleRef: v })}>
   824→              <SelectTrigger className="h-9 text-xs"><SelectValue placeholder="Référence biblique" /></SelectTrigger>
   825→              <SelectContent>
   826→                <div className="max-h-48 overflow-y-auto">
   827→                  {BIBLE_BOOKS.map((b) => (
   828→                    <SelectItem key={b.name} value={b.name} className="text-xs">{b.name}</SelectItem>
   829→                  ))}
   830→                </div>
   831→              </SelectContent>
   832→            </Select>
   833→          )}
   834→          {noteContext.bookId && books.length > 0 && !noteContext.bibleRef && (
   835→            <Select value={noteContext.bookId} onValueChange={(v) => setNoteContext({ bookId: v })}>
   836→              <SelectTrigger className="h-9 text-xs"><SelectValue placeholder="Choisir un livre" /></SelectTrigger>
   837→              <SelectContent>
   838→                {books.map((b) => (
   839→                  <SelectItem key={b.id} value={b.id} className="text-xs">{b.title}</SelectItem>
   840→                ))}
   841→              </SelectContent>
   842→            </Select>
   843→          )}
   844→          <Textarea
   845→            value={noteText}
   846→            onChange={(e) => setNoteText(e.target.value)}
   847→            placeholder="Écrivez votre note ici..."
   848→            className="min-h-[100px] text-sm"
   849→            autoFocus
   850→          />
   851→          <Button
   852→            onClick={() => {
   853→              if (!noteText.trim()) return;
   854→              // Random position to avoid overlap
   855→              const x = 10 + Math.random() * 60;
   856→              const y = 10 + Math.random() * 60;
   857→              addNote.mutate({ ...noteContext, content: noteText, positionX: x, positionY: y });
   858→            }}
   859→            disabled={!noteText.trim()}
   860→            className="w-full bg-[var(--theme-primary)] hover:bg-[var(--theme-primary-hover)] text-white text-xs"
   861→          >
   862→            <Plus className="h-3 w-3 mr-1" /> Ajouter la note
   863→          </Button>
   864→        </DialogContent>
   865→      </Dialog>
   866→
   867→      {/* PDF Viewer Dialog */}
   868→      <PdfViewerDialog
   869→        open={pdfViewerOpen}
   870→        onOpenChange={setPdfViewerOpen}
   871→        pdfUrl={pdfViewerUrl}
   872→        title={pdfViewerTitle}
   873→      />
   874→    </div>
   875→  );
   876→}
