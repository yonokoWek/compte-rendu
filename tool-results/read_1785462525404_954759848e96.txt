     1→'use client';
     2→
     3→import React, { useState } from 'react';
     4→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
     5→import { format, eachDayOfInterval, getDay, differenceInCalendarDays } from 'date-fns';
     6→import { fr } from 'date-fns/locale';
     7→import { useAppStore, DAY_NAMES_SHORT, formatMinutes, getDaysInRange, getWeeksInRange } from '@/store/app-store';
     8→import { Card, CardContent } from '@/components/ui/card';
     9→import { Button } from '@/components/ui/button';
    10→import { Input } from '@/components/ui/input';
    11→import { Skeleton } from '@/components/ui/skeleton';
    12→import {
    13→  Select,
    14→  SelectContent,
    15→  SelectItem,
    16→  SelectTrigger,
    17→  SelectValue,
    18→} from '@/components/ui/select';
    19→import {
    20→  Dialog,
    21→  DialogContent,
    22→  DialogHeader,
    23→  DialogTitle,
    24→  DialogDescription,
    25→} from '@/components/ui/dialog';
    26→import {
    27→  DropdownMenu,
    28→  DropdownMenuContent,
    29→  DropdownMenuItem,
    30→  DropdownMenuTrigger,
    31→  DropdownMenuSeparator,
    32→} from '@/components/ui/dropdown-menu';
    33→import {
    34→  ChevronLeft,
    35→  ChevronRight,
    36→  Download,
    37→  User,
    38→  CalendarDays,
    39→  Settings,
    40→  Plus,
    41→  Trash2,
    42→  FolderPlus,
    43→  GripVertical,
    44→  FolderOpen,
    45→  X,
    46→  ArrowRight,
    47→  MoreVertical,
    48→} from 'lucide-react';
    49→import { toast } from 'sonner';
    50→
    51→interface Category {
    52→  id: string;
    53→  name: string;
    54→  unit: string;
    55→  sortOrder: number;
    56→  isPersonal: boolean;
    57→  icon: string;
    58→  groupId: string | null;
    59→  group: { id: string; name: string } | null;
    60→}
    61→
    62→interface Group {
    63→  id: string;
    64→  name: string;
    65→  sortOrder: number;
    66→  categories: Category[];
    67→}
    68→
    69→interface DailyEntry {
    70→  id: string;
    71→  date: string;
    72→  categoryId: string;
    73→  value: number;
    74→  category: Category;
    75→}
    76→
    77→interface UserProfile {
    78→  id: string;
    79→  firstName: string;
    80→  lastName: string;
    81→  assembly: string;
    82→  mentor: string;
    83→}
    84→
    85→export default function CompteRenduTab() {
    86→  const period = useAppStore((s) => s.period);
    87→  const periodType = useAppStore((s) => s.periodType);
    88→  const setPeriodType = useAppStore((s) => s.setPeriodType);
    89→  const nextPeriod = useAppStore((s) => s.nextPeriod);
    90→  const prevPeriod = useAppStore((s) => s.prevPeriod);
    91→  const goToCurrentPeriod = useAppStore((s) => s.goToCurrentPeriod);
    92→  const setProfileDialogOpen = useAppStore((s) => s.setProfileDialogOpen);
    93→  const queryClient = useQueryClient();
    94→
    95→  // State for management dialog
    96→  const [manageOpen, setManageOpen] = useState(false);
    97→  const [newGroupName, setNewGroupName] = useState('');
    98→  const [newCatName, setNewCatName] = useState('');
    99→  const [newCatUnit, setNewCatUnit] = useState('minutes');
   100→  const [newCatPersonal, setNewCatPersonal] = useState(true);
   101→  const [assigningCat, setAssigningCat] = useState<Category | null>(null);
   102→
   103→  // Fetch categories with groups
   104→  const { data: catData } = useQuery<{
   105→    categories: Category[];
   106→    groups: Group[];
   107→  }>({
   108→    queryKey: ['categories'],
   109→    queryFn: () => fetch('/api/categories').then((r) => r.json()),
   110→  });
   111→  const categories = catData?.categories || [];
   112→  const groups = catData?.groups || [];
   113→
   114→  // Fetch entries
   115→  const { data: entries = [], isLoading: entriesLoading } = useQuery<DailyEntry[]>({
   116→    queryKey: ['entries', period.startDate, period.endDate],
   117→    queryFn: () =>
   118→      fetch(
   119→        `/api/entries?startDate=${format(period.startDate, 'yyyy-MM-dd')}&endDate=${format(period.endDate, 'yyyy-MM-dd')}`
   120→      ).then((r) => r.json()),
   121→  });
   122→
   123→  // Fetch profile
   124→  const { data: profile } = useQuery<UserProfile>({
   125→    queryKey: ['profile'],
   126→    queryFn: () => fetch('/api/profile').then((r) => r.json()),
   127→  });
   128→
   129→  // Save entry mutation
   130→  const saveEntry = useMutation({
   131→    mutationFn: (data: { date: string; categoryId: string; value: number }) =>
   132→      fetch('/api/entries', {
   133→        method: 'POST',
   134→        headers: { 'Content-Type': 'application/json' },
   135→        body: JSON.stringify(data),
   136→      }).then((r) => r.json()),
   137→    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['entries'] }),
   138→  });
   139→
   140→  // Category mutations
   141→  const addCategory = useMutation({
   142→    mutationFn: (data: { type: 'category'; name: string; unit: string; isPersonal: boolean; groupId?: string }) =>
   143→      fetch('/api/categories', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) }).then((r) => r.json()),
   144→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: ['categories'] }); setNewCatName(''); },
   145→  });
   146→
   147→  const deleteCategory = useMutation({
   148→    mutationFn: (id: string) => fetch(`/api/categories?type=category&id=${id}`, { method: 'DELETE' }),
   149→    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['categories'] }),
   150→  });
   151→
   152→  const addGroup = useMutation({
   153→    mutationFn: (name: string) =>
   154→      fetch('/api/categories', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ type: 'group', name }) }).then((r) => r.json()),
   155→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: ['categories'] }); setNewGroupName(''); toast.success('Groupe créé'); },
   156→  });
   157→
   158→  const deleteGroup = useMutation({
   159→    mutationFn: (id: string) => fetch(`/api/categories?type=group&id=${id}`, { method: 'DELETE' }),
   160→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: ['categories'] }); toast.success('Groupe supprimé'); },
   161→  });
   162→
   163→  const assignToGroup = useMutation({
   164→    mutationFn: (data: { type: 'assign'; categoryId: string; groupId: string | null }) =>
   165→      fetch('/api/categories', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) }).then((r) => r.json()),
   166→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: ['categories'] }); setAssigningCat(null); toast.success('Activité déplacée'); },
   167→  });
   168→
   169→  // Calculate columns
   170→  const totalDays = differenceInCalendarDays(period.endDate, period.startDate) + 1;
   171→  const isSingleWeek = totalDays <= 7;
   172→
   173→  let columns: { key: string; label: string; date: string }[] = [];
   174→  if (isSingleWeek) {
   175→    columns = getDaysInRange(period.startDate, period.endDate).map((d) => {
   176→      const dayIndex = getDay(d);
   177→      return { key: format(d, 'yyyy-MM-dd'), label: DAY_NAMES_SHORT[dayIndex === 0 ? 6 : dayIndex - 1], date: format(d, 'yyyy-MM-dd') };
   178→    });
   179→  } else {
   180→    columns = getWeeksInRange(period.startDate, period.endDate).map((w, i) => ({
   181→      key: `week-${i}`,
   182→      label: `Sem ${i + 1}`,
   183→      date: `${format(w.start, 'yyyy-MM-dd')}_${format(w.end, 'yyyy-MM-dd')}`,
   184→    }));
   185→  }
   186→
   187→  const entryMap: Record<string, number> = {};
   188→  for (const e of entries) entryMap[`${e.date}_${e.categoryId}`] = e.value;
   189→
   190→  const getCellValue = (categoryId: string, col: typeof columns[0]) => {
   191→    if (isSingleWeek) return entryMap[`${col.date}_${categoryId}`] || 0;
   192→    const [wStart, wEnd] = col.date.split('_');
   193→    let total = 0;
   194→    for (const day of eachDayOfInterval({ start: new Date(wStart), end: new Date(wEnd) })) {
   195→      total += entryMap[`${format(day, 'yyyy-MM-dd')}_${categoryId}`] || 0;
   196→    }
   197→    return total;
   198→  };
   199→
   200→  const getRowTotal = (catId: string) => columns.reduce((s, col) => s + getCellValue(catId, col), 0);
   201→
   202→  const personalCategories = categories.filter((c) => c.isPersonal && c.unit === 'minutes');
   203→  const getPersonalTotal = (col?: typeof columns[0]) => {
   204→    if (col) return personalCategories.reduce((s, cat) => s + getCellValue(cat.id, col), 0);
   205→    return columns.reduce((s, col) => s + personalCategories.reduce((s2, cat) => s2 + getCellValue(cat.id, col), 0), 0);
   206→  };
   207→
   208→  const handleCellChange = (categoryId: string, columnKey: string, value: string) => {
   209→    const numValue = parseInt(value) || 0;
   210→    if (isSingleWeek) {
   211→      saveEntry.mutate({ date: columnKey, categoryId, value: numValue });
   212→    } else {
   213→      const [wStart] = columnKey.split('_');
   214→      saveEntry.mutate({ date: wStart, categoryId, value: numValue });
   215→    }
   216→  };
   217→
   218→  // PDF generation
   219→  const [pdfLoading, setPdfLoading] = useState(false);
   220→  const handleExportPDF = async () => {
   221→    setPdfLoading(true);
   222→    try {
   223→      const reportRes = await fetch('/api/report', {
   224→        method: 'POST',
   225→        headers: { 'Content-Type': 'application/json' },
   226→        body: JSON.stringify({ startDate: format(period.startDate, 'yyyy-MM-dd'), endDate: format(period.endDate, 'yyyy-MM-dd') }),
   227→      });
   228→      const reportData = await reportRes.json();
   229→      const pdfRes = await fetch('/api/generate-pdf', {
   230→        method: 'POST',
   231→        headers: { 'Content-Type': 'application/json' },
   232→        body: JSON.stringify({ html: reportData.html }),
   233→      });
   234→      const pdfData = await pdfRes.json();
   235→      const link = document.createElement('a');
   236→      link.href = pdfData.pdf;
   237→      link.download = `compte-rendu-${format(period.startDate, 'yyyy-MM-dd')}.pdf`;
   238→      link.click();
   239→      toast.success('PDF exporté');
   240→    } catch {
   241→      toast.error('Erreur PDF');
   242→    } finally {
   243→      setPdfLoading(false);
   244→    }
   245→  };
   246→
   247→  const displayName = profile ? `${profile.firstName} ${profile.lastName}`.trim() : 'Mon Compte';
   248→
   249→  // Group categories for display
   250→  const groupedDisplay = new Map<string, Category[]>();
   251→  const ungrouped: Category[] = [];
   252→  for (const cat of categories) {
   253→    if (cat.groupId && cat.group) {
   254→      if (!groupedDisplay.has(cat.groupId)) groupedDisplay.set(cat.groupId, []);
   255→      groupedDisplay.get(cat.groupId)!.push(cat);
   256→    } else {
   257→      ungrouped.push(cat);
   258→    }
   259→  }
   260→
   261→  return (
   262→    <div className="max-w-6xl mx-auto p-4 space-y-4">
   263→      {/* Header */}
   264→      <div className="flex items-center justify-between">
   265→        <div className="flex items-center gap-2">
   266→          <div className="bg-[var(--theme-primary-light)] p-2 rounded-full">
   267→            <CalendarDays className="h-5 w-5 text-[var(--theme-primary)]" />
   268→          </div>
   269→          <div>
   270→            <h1 className="text-lg font-bold text-gray-900">Compte Rendu</h1>
   271→            <p className="text-xs text-gray-500">{displayName} {profile?.assembly ? `• ${profile.assembly}` : ''}</p>
   272→          </div>
   273→        </div>
   274→        <div className="flex items-center gap-1">
   275→          <Button variant="ghost" size="icon" onClick={() => setManageOpen(true)}>
   276→            <Settings className="h-4 w-4" />
   277→          </Button>
   278→          <Button variant="ghost" size="icon" onClick={() => setProfileDialogOpen(true)}>
   279→            <User className="h-4 w-4" />
   280→          </Button>
   281→          <Button variant="outline" size="sm" onClick={handleExportPDF} disabled={pdfLoading} className="gap-1 text-[var(--theme-primary)] border-[var(--theme-primary)] hover:bg-[var(--theme-primary-light)]">
   282→            <Download className="h-4 w-4" />
   283→            <span className="hidden sm:inline">PDF</span>
   284→          </Button>
   285→        </div>
   286→      </div>
   287→
   288→      {/* Period selector */}
   289→      <Card className="border-[var(--theme-primary)]">
   290→        <CardContent className="p-3">
   291→          <div className="flex items-center justify-between gap-2">
   292→            <Button variant="ghost" size="icon" className="shrink-0" onClick={prevPeriod}>
   293→              <ChevronLeft className="h-4 w-4" />
   294→            </Button>
   295→            <div className="flex-1 flex items-center justify-center gap-2">
   296→              <Select value={periodType} onValueChange={(v) => setPeriodType(v as 'week' | 'month' | 'year')}>
   297→                <SelectTrigger className="w-28 h-8 text-xs"><SelectValue /></SelectTrigger>
   298→                <SelectContent>
   299→                  <SelectItem value="week">Semaine</SelectItem>
   300→                  <SelectItem value="month">Mois</SelectItem>
   301→                  <SelectItem value="year">Année</SelectItem>
   302→                </SelectContent>
   303→              </Select>
   304→              <span className="text-sm font-medium text-gray-800 truncate">{period.label}</span>
   305→            </div>
   306→            <div className="flex items-center gap-1">
   307→              <Button variant="ghost" size="sm" className="text-xs h-8 px-2" onClick={goToCurrentPeriod}>Auj.</Button>
   308→              <Button variant="ghost" size="icon" className="shrink-0" onClick={nextPeriod}>
   309→                <ChevronRight className="h-4 w-4" />
   310→              </Button>
   311→            </div>
   312→          </div>
   313→        </CardContent>
   314→      </Card>
   315→
   316→      {/* Main data table */}
   317→      <Card>
   318→        <CardContent className="p-0">
   319→          <div className="overflow-x-auto">
   320→            <table className="w-full text-xs border-collapse">
   321→              <thead>
   322→                <tr>
   323→                  <th className="sticky left-0 bg-[var(--theme-primary)] text-white px-2 py-2 text-left font-semibold text-xs min-w-[130px] z-10 rounded-tl-lg">Activité</th>
   324→                  <th className="bg-[var(--theme-primary)] text-white px-1 py-2 text-center font-semibold text-xs w-10">{isSingleWeek ? 'Min' : ''}</th>
   325→                  {columns.map((col) => (
   326→                    <th key={col.key} className="bg-[var(--theme-primary)] text-white px-2 py-2 text-center font-semibold text-xs min-w-[55px]">{col.label}</th>
   327→                  ))}
   328→                  <th className="bg-[var(--theme-primary)] text-white px-2 py-2 text-center font-semibold text-xs min-w-[55px] rounded-tr-lg">Total</th>
   329→                </tr>
   330→              </thead>
   331→              <tbody>
   332→                {entriesLoading ? (
   333→                  Array.from({ length: 8 }).map((_, i) => (
   334→                    <tr key={i}>
   335→                      <td className="bg-[var(--theme-primary-light)] px-2 py-2"><Skeleton className="h-4 w-24" /></td>
   336→                      {columns.map((_, j) => <td key={j} className="px-2 py-2"><Skeleton className="h-6 w-8 mx-auto" /></td>)}
   337→                    </tr>
   338→                  ))
   339→                ) : (
   340→                  <>
   341→                    {/* Grouped rows with group headers */}
   342→                    {Array.from(groupedDisplay.entries()).map(([gId, cats]) => {
   343→                      const groupName = cats[0]?.group?.name || '';
   344→                      return (
   345→                        <React.Fragment key={`group-${gId}`}>
   346→                          <tr>
   347→                            <td colSpan={columns.length + 3} className="bg-[var(--theme-primary)] text-white px-3 py-1 text-left text-xs font-bold">
   348→                              <FolderOpen className="h-3 w-3 inline mr-1" />
   349→                              {groupName}
   350→                            </td>
   351→                          </tr>
   352→                          {cats.map((cat, rowIndex) => (
   353→                            <tr key={cat.id} className={rowIndex % 2 === 0 ? 'bg-white' : 'bg-[var(--theme-primary-light)]/50'}>
   354→                              <td className="sticky left-0 px-2 py-1.5 text-left font-medium text-xs bg-inherit z-10 whitespace-nowrap border-r border-[var(--theme-primary-light)]">
   355→                                <span className="text-[10px] text-[var(--theme-primary)]">●</span> {cat.name}
   356→                              </td>
   357→                              <td className="px-1 py-1.5 text-center text-[10px] text-gray-400 bg-[var(--theme-primary-light)]/50">
   358→                                {cat.unit === 'minutes' ? 'min' : 'Part.'}
   359→                              </td>
   360→                              {columns.map((col) => {
   361→                                const val = getCellValue(cat.id, col);
   362→                                return (
   363→                                  <td key={col.key} className="px-1 py-1 text-center">
   364→                                    <input type="number" min="0" value={val || ''} onChange={(e) => handleCellChange(cat.id, col.key, e.target.value)}
   365→                                      placeholder="0"
   366→                                      className="w-12 h-7 text-center text-xs bg-transparent border border-transparent rounded hover:border-[var(--theme-primary)] focus:border-[var(--theme-primary)] focus:bg-[var(--theme-primary-light)] focus:outline-none transition-colors" />
   367→                                  </td>
   368→                                );
   369→                              })}
   370→                              <td className="px-2 py-1.5 text-center font-bold text-xs bg-[var(--theme-primary-light)]/80">
   371→                                {cat.unit === 'minutes' ? formatMinutes(getRowTotal(cat.id)) : getRowTotal(cat.id) || ''}
   372→                              </td>
   373→                            </tr>
   374→                          ))}
   375→                        </React.Fragment>
   376→                      );
   377→                    })}
   378→
   379→                    {/* Ungrouped rows */}
   380→                    {ungrouped.length > 0 && (
   381→                      <>
   382→                        <tr>
   383→                          <td colSpan={columns.length + 3} className="bg-gray-200 text-gray-600 px-3 py-1 text-left text-xs font-semibold">
   384→                            Sans groupe
   385→                          </td>
   386→                        </tr>
   387→                        {ungrouped.map((cat, rowIndex) => (
   388→                          <tr key={cat.id} className={rowIndex % 2 === 0 ? 'bg-white' : 'bg-[var(--theme-primary-light)]/50'}>
   389→                            <td className="sticky left-0 px-2 py-1.5 text-left font-medium text-xs bg-inherit z-10 whitespace-nowrap border-r border-[var(--theme-primary-light)]">
   390→                              <span className="text-[10px] text-gray-400">●</span> {cat.name}
   391→                            </td>
   392→                            <td className="px-1 py-1.5 text-center text-[10px] text-gray-400 bg-[var(--theme-primary-light)]/50">
   393→                              {cat.unit === 'minutes' ? 'min' : 'Part.'}
   394→                            </td>
   395→                            {columns.map((col) => {
   396→                              const val = getCellValue(cat.id, col);
   397→                              return (
   398→                                <td key={col.key} className="px-1 py-1 text-center">
   399→                                  <input type="number" min="0" value={val || ''} onChange={(e) => handleCellChange(cat.id, col.key, e.target.value)}
   400→                                    placeholder="0"
   401→                                    className="w-12 h-7 text-center text-xs bg-transparent border border-transparent rounded hover:border-[var(--theme-primary)] focus:border-[var(--theme-primary)] focus:bg-[var(--theme-primary-light)] focus:outline-none transition-colors" />
   402→                                </td>
   403→                              );
   404→                            })}
   405→                            <td className="px-2 py-1.5 text-center font-bold text-xs bg-[var(--theme-primary-light)]/80">
   406→                              {cat.unit === 'minutes' ? formatMinutes(getRowTotal(cat.id)) : getRowTotal(cat.id) || ''}
   407→                            </td>
   408→                          </tr>
   409→                        ))}
   410→                      </>
   411→                    )}
   412→
   413→                    {/* Personal total row */}
   414→                    <tr className="bg-[var(--theme-primary-hover)]">
   415→                      <td colSpan={2} className="sticky left-0 px-2 py-2 text-left font-bold text-xs text-white z-10">
   416→                        Temps seul avec le Seigneur
   417→                      </td>
   418→                      {columns.map((col) => (
   419→                        <td key={col.key} className="px-2 py-2 text-center font-bold text-xs text-white">
   420→                          {formatMinutes(getPersonalTotal(col))}
   421→                        </td>
   422→                      ))}
   423→                      <td className="px-2 py-2 text-center font-bold text-sm text-white">
   424→                        {formatMinutes(getPersonalTotal())}
   425→                      </td>
   426→                    </tr>
   427→                  </>
   428→                )}
   429→              </tbody>
   430→            </table>
   431→          </div>
   432→        </CardContent>
   433→      </Card>
   434→
   435→      {/* Management Dialog */}
   436→      <Dialog open={manageOpen} onOpenChange={setManageOpen}>
   437→        <DialogContent className="sm:max-w-lg max-h-[80vh] overflow-y-auto">
   438→          <DialogHeader>
   439→            <DialogTitle className="flex items-center gap-2">
   440→              <Settings className="h-5 w-5 text-[var(--theme-primary)]" />
   441→              Gérer les activités
   442→            </DialogTitle>
   443→            <DialogDescription>
   444→              Créez des groupes, ajoutez/supprimez des activités et assignez-les aux groupes.
   445→              Seules les activités groupées apparaîtront sur le PDF.
   446→            </DialogDescription>
   447→          </DialogHeader>
   448→
   449→          <div className="space-y-4 py-2">
   450→            {/* Create group */}
   451→            <Card className="border-[var(--theme-primary)]">
   452→              <CardContent className="p-3 space-y-2">
   453→                <h4 className="text-xs font-semibold text-gray-700">Créer un groupe</h4>
   454→                <div className="flex gap-2">
   455→                  <Input placeholder="Nom du groupe" value={newGroupName} onChange={(e) => setNewGroupName(e.target.value)} className="text-sm flex-1" />
   456→                  <Button onClick={() => { if (newGroupName.trim()) addGroup.mutate(newGroupName); }} disabled={!newGroupName.trim() || addGroup.isPending} size="sm" className="bg-[var(--theme-primary)] hover:bg-[var(--theme-primary-hover)] text-white text-xs">
   457→                    <FolderPlus className="h-3 w-3 mr-1" /> Créer
   458→                  </Button>
   459→                </div>
   460→              </CardContent>
   461→            </Card>
   462→
   463→            {/* Existing groups */}
   464→            {groups.length > 0 && (
   465→              <div className="space-y-2">
   466→                <h4 className="text-xs font-semibold text-gray-700">Groupes existants</h4>
   467→                {groups.map((group) => (
   468→                  <Card key={group.id} className="border-l-4 border-l-[var(--theme-primary)]">
   469→                    <CardContent className="p-3">
