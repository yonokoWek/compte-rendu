    88→    88→  const setPeriodType = useAppStore((s) => s.setPeriodType);

    93→    93→  const queryClient = useQueryClient();
    94→    94→
    95→    95→  // State for management dialog

    97→    97→  const [newGroupName, setNewGroupName] = useState('');
    98→    98→  const [newCatName, setNewCatName] = useState('');

   102→   102→

   105→   105→    categories: Category[];

   107→   107→  }>({

   109→   109→    queryFn: () => fetch('/api/categories').then((r) => r.json()),

   112→   112→  const groups = catData?.groups || [];
   113→   113→
   114→   114→  // Fetch entries
   115→   115→  const { data: entries = [], isLoading: entriesLoading } = useQuery<DailyEntry[]>({
   116→   116→    queryKey: ['entries', period.startDate, period.endDate],
   117→   117→    queryFn: () =>
   118→   118→      fetch(
   119→   119→        `/api/entries?startDate=${format(period.startDate, 'yyyy-MM-dd')}&endDate=${format(period.endDate, 'yyyy-MM-dd')}`
   120→   120→      ).then((r) => r.json()),
   121→   121→  });
   122→   122→
   123→   123→  // Fetch profile
   124→   124→  const { data: profile } = useQuery<UserProfile>({
   125→   125→    queryKey: ['profile'],
   126→   126→    queryFn: () => fetch('/api/profile').then((r) => r.json()),
   127→   127→  });

   131→   131→    mutationFn: (data: { date: string; categoryId: string; value: number }) =>
   132→   132→      fetch('/api/entries', {
   133→   133→        method: 'POST',
   134→   134→        headers: { 'Content-Type': 'application/json' },
   135→   135→        body: JSON.stringify(data),
   136→   136→      }).then((r) => r.json()),
   137→   137→    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['entries'] }),
   138→   138→  });
   139→   139→
   140→   140→  // Category mutations
   141→   141→  const addCategory = useMutation({
   142→   142→    mutationFn: (data: { type: 'category'; name: string; unit: string; isPersonal: boolean; groupId?: string }) =>
   143→   143→      fetch('/api/categories', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) }).then((r) => r.json()),
   144→   144→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: ['categories'] }); setNewCatName(''); },
   145→   145→  });
   146→   146→
   147→   147→  const deleteCategory = useMutation({
   148→   148→    mutationFn: (id: string) => fetch(`/api/categories?type=category&id=${id}`, { method: 'DELETE' }),
   149→   149→    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['categories'] }),
   150→   150→  });
   151→   151→
   152→   152→  const addGroup = useMutation({
   153→   153→    mutationFn: (name: string) =>
   154→   154→      fetch('/api/categories', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ type: 'group', name }) }).then((r) => r.json()),
   155→   155→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: ['categories'] }); setNewGroupName(''); toast.success('Groupe créé'); },
   156→   156→  });
   157→   157→
   158→   158→  const deleteGroup = useMutation({
   159→   159→    mutationFn: (id: string) => fetch(`/api/categories?type=group&id=${id}`, { method: 'DELETE' }),
   160→   160→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: ['categories'] }); toast.success('Groupe supprimé'); },
   161→   161→  });
   162→   162→
   163→   163→  const assignToGroup = useMutation({
   164→   164→    mutationFn: (data: { type: 'assign'; categoryId: string; groupId: string | null }) =>
   165→   165→      fetch('/api/categories', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) }).then((r) => r.json()),
   166→   166→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: ['categories'] }); setAssigningCat(null); toast.success('Activité déplacée'); },
   167→   167→  });
   168→   168→
   169→   169→  // Calculate columns
   170→   170→  const totalDays = differenceInCalendarDays(period.endDate, period.startDate) + 1;
   171→   171→  const isSingleWeek = totalDays <= 7;

   173→   173→  let columns: { key: string; label: string; date: string }[] = [];

   175→   175→    columns = getDaysInRange(period.startDate, period.endDate).map((d) => {
   176→   176→      const dayIndex = getDay(d);
   177→   177→      return { key: format(d, 'yyyy-MM-dd'), label: DAY_NAMES_SHORT[dayIndex === 0 ? 6 : dayIndex - 1], date: format(d, 'yyyy-MM-dd') };

   182→   182→      label: `Sem ${i + 1}`,
   183→   183→      date: `${format(w.start, 'yyyy-MM-dd')}_${format(w.end, 'yyyy-MM-dd')}`,

   192→   192→    const [wStart, wEnd] = col.date.split('_');
   193→   193→    let total = 0;

   195→   195→      total += entryMap[`${format(day, 'yyyy-MM-dd')}_${categoryId}`] || 0;
   196→   196→    }
   197→   197→    return total;
   198→   198→  };

   200→   200→  const getRowTotal = (catId: string) => columns.reduce((s, col) => s + getCellValue(catId, col), 0);

   202→   202→  const personalCategories = categories.filter((c) => c.isPersonal && c.unit === 'minutes');

   204→   204→    if (col) return personalCategories.reduce((s, cat) => s + getCellValue(cat.id, col), 0);
   205→   205→    return columns.reduce((s, col) => s + personalCategories.reduce((s2, cat) => s2 + getCellValue(cat.id, col), 0), 0);
   206→   206→  };

   208→   208→  const handleCellChange = (categoryId: string, columnKey: string, value: string) => {

   210→   210→    if (isSingleWeek) {
   211→   211→      saveEntry.mutate({ date: columnKey, categoryId, value: numValue });
   212→   212→    } else {
   213→   213→      const [wStart] = columnKey.split('_');
   214→   214→      saveEntry.mutate({ date: wStart, categoryId, value: numValue });
   215→   215→    }
   216→   216→  };

   218→   218→  // PDF generation
   219→   219→  const [pdfLoading, setPdfLoading] = useState(false);

   224→   224→        method: 'POST',
   225→   225→        headers: { 'Content-Type': 'application/json' },

   230→   230→        method: 'POST',

   232→   232→        body: JSON.stringify({ html: reportData.html }),
   233→   233→      });
   234→   234→      const pdfData = await pdfRes.json();

   237→   237→      link.download = `compte-rendu-${format(period.startDate, 'yyyy-MM-dd')}.pdf`;
   238→   238→      link.click();
   239→   239→      toast.success('PDF exporté');
   240→   240→    } catch {
   241→   241→      toast.error('Erreur PDF');
   242→   242→    } finally {
   243→   243→      setPdfLoading(false);
   244→   244→    }
   245→   245→  };
   246→   246→
   247→   247→  const displayName = profile ? `${profile.firstName} ${profile.lastName}`.trim() : 'Mon Compte';
   248→   248→
   249→   249→  // Group categories for display
   250→   250→  const groupedDisplay = new Map<string, Category[]>();
   251→   251→  const ungrouped: Category[] = [];
   252→   252→  for (const cat of categories) {
   253→   253→    if (cat.groupId && cat.group) {
   254→   254→      if (!groupedDisplay.has(cat.groupId)) groupedDisplay.set(cat.groupId, []);
   255→   255→      groupedDisplay.get(cat.groupId)!.push(cat);
   256→   256→    } else {
   257→   257→      ungrouped.push(cat);
   258→   258→    }
   259→   259→  }

   265→   265→        <div className="flex items-center gap-2">
   266→   266→          <div className="bg-orange-100 p-2 rounded-full">
   267→   267→            <CalendarDays className="h-5 w-5 text-orange-600" />

   276→   276→            <Settings className="h-4 w-4" />
   277→   277→          </Button>

   279→   279→            <User className="h-4 w-4" />
   280→   280→          </Button>
   281→   281→          <Button variant="outline" size="sm" onClick={handleExportPDF} disabled={pdfLoading} className="gap-1 text-orange-600 border-orange-200 hover:bg-orange-50">
   282→   282→            <Download className="h-4 w-4" />
   283→   283→            <span className="hidden sm:inline">PDF</span>
   284→   284→          </Button>
   285→   285→        </div>
   286→   286→      </div>
   287→   287→
   288→   288→      {/* Period selector */}

   299→   299→                  <SelectItem value="week">Semaine</SelectItem>
   300→   300→                  <SelectItem value="month">Mois</SelectItem>

   302→   302→                </SelectContent>

   304→   304→              <span className="text-sm font-medium text-gray-800 truncate">{period.label}</span>

   307→   307→              <Button variant="ghost" size="sm" className="text-xs h-8 px-2" onClick={goToCurrentPeriod}>Auj.</Button>

   330→   330→              </thead>
   331→   331→              <tbody>
   332→   332→                {entriesLoading ? (
   333→   333→                  Array.from({ length: 8 }).map((_, i) => (

   352→   352→                          {cats.map((cat, rowIndex) => (

   355→   355→                                <span className="text-[10px] text-orange-500">●</span> {cat.name}

   367→   367→                                  </td>
   368→   368→                                );
   369→   369→                              })}

   383→   383→                          <td colSpan={columns.length + 3} className="bg-gray-200 text-gray-600 px-3 py-1 text-left text-xs font-semibold">
   384→   384→                            Sans groupe
   385→   385→                          </td>
   386→   386→                        </tr>
   387→   387→                        {ungrouped.map((cat, rowIndex) => (
   388→   388→                          <tr key={cat.id} className={rowIndex % 2 === 0 ? 'bg-white' : 'bg-orange-50/50'}>
   389→   389→                            <td className="sticky left-0 px-2 py-1.5 text-left font-medium text-xs bg-inherit z-10 whitespace-nowrap border-r border-orange-100">

   391→   391→                            </td>
   392→   392→                            <td className="px-1 py-1.5 text-center text-[10px] text-gray-400 bg-orange-100/50">

   401→   401→                                    className="w-12 h-7 text-center text-xs bg-transparent border border-transparent rounded hover:border-orange-300 focus:border-orange-400 focus:bg-orange-50 focus:outline-none transition-colors" />
