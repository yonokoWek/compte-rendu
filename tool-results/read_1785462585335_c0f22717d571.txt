   120→// Floating Note Component with proper drag handling
   121→function FloatingNote({ note, onUpdate, onDelete }: {
   122→  note: ReadingNote;
   123→  onUpdate: (id: string, content: string, positionX: number, positionY: number) => void;
   124→  onDelete: (id: string) => void;
   125→}) {
   126→  const noteRef = useRef<HTMLDivElement>(null);
   127→  const [editing, setEditing] = useState(false);
   128→  const isDragging = useRef(false);
   129→  const startPos = useRef({ x: 0, y: 0 });
   130→  const startMousePos = useRef({ x: 0, y: 0 });
   131→  const [editText, setEditText] = useState(note.content);
   132→
   133→  const startEditing = useCallback(() => {
   134→    setEditText(note.content);
   135→    setEditing(true);
   136→  }, [note.content]);
   137→
   138→  const handleMouseDown = useCallback((e: React.MouseEvent) => {
   139→    if (editing) return; // Don't drag while editing
   140→    e.preventDefault();
   141→    isDragging.current = false;
   142→    startPos.current = { x: note.positionX, y: note.positionY };
   143→    startMousePos.current = { x: e.clientX, y: e.clientY };
   144→
   145→    const handleMove = (me: MouseEvent) => {
   146→      const dx = me.clientX - startMousePos.current.x;
   147→      const dy = me.clientY - startMousePos.current.y;
   148→      if (Math.abs(dx) > 3 || Math.abs(dy) > 3) {
   149→        isDragging.current = true;
   150→      }
   151→      if (!isDragging.current || !noteRef.current) return;
   152→      const parent = noteRef.current.parentElement;
   153→      if (!parent) return;
   154→      const rect = parent.getBoundingClientRect();
   155→      const newX = Math.max(0, Math.min(95, ((startPos.current.x / 100) * rect.width + dx) / rect.width * 100));
   156→      const newY = Math.max(0, Math.min(90, ((startPos.current.y / 100) * rect.height + dy) / rect.height * 100));
   157→      noteRef.current.style.left = `${newX}%`;
   158→      noteRef.current.style.top = `${newY}%`;
   159→    };
   160→
   161→    const handleUp = (me: MouseEvent) => {
   162→      document.removeEventListener('mousemove', handleMove);
   163→      document.removeEventListener('mouseup', handleUp);
   164→      if (isDragging.current && noteRef.current) {
   165→        const left = parseFloat(noteRef.current.style.left) || note.positionX;
   166→        const top = parseFloat(noteRef.current.style.top) || note.positionY;
   167→        onUpdate(note.id, note.content, left, top);
   168→      }
   169→      isDragging.current = false;
   170→    };
   171→
   172→    document.addEventListener('mousemove', handleMove);
   173→    document.addEventListener('mouseup', handleUp);
   174→  }, [note.id, note.content, note.positionX, note.positionY, editing, onUpdate]);
   175→
   176→  // Touch support for mobile
   177→  const handleTouchStart = useCallback((e: React.TouchEvent) => {
   178→    if (editing) return;
   179→    const touch = e.touches[0];
   180→    isDragging.current = false;
   181→    startPos.current = { x: note.positionX, y: note.positionY };
   182→    startMousePos.current = { x: touch.clientX, y: touch.clientY };
   183→
   184→    const handleTouchMove = (te: TouchEvent) => {
   185→      const t = te.touches[0];
   186→      const dx = t.clientX - startMousePos.current.x;
   187→      const dy = t.clientY - startMousePos.current.y;
   188→      if (Math.abs(dx) > 3 || Math.abs(dy) > 3) {
   189→        isDragging.current = true;
   190→      }
   191→      if (!isDragging.current || !noteRef.current) return;
   192→      const parent = noteRef.current.parentElement;
   193→      if (!parent) return;
   194→      const rect = parent.getBoundingClientRect();
   195→      const newX = Math.max(0, Math.min(95, ((startPos.current.x / 100) * rect.width + dx) / rect.width * 100));
   196→      const newY = Math.max(0, Math.min(90, ((startPos.current.y / 100) * rect.height + dy) / rect.height * 100));
   197→      noteRef.current.style.left = `${newX}%`;
   198→      noteRef.current.style.top = `${newY}%`;
   199→    };
   200→
   201→    const handleTouchEnd = () => {
   202→      document.removeEventListener('touchmove', handleTouchMove);
   203→      document.removeEventListener('touchend', handleTouchEnd);
   204→      if (isDragging.current && noteRef.current) {
   205→        const left = parseFloat(noteRef.current.style.left) || note.positionX;
   206→        const top = parseFloat(noteRef.current.style.top) || note.positionY;
   207→        onUpdate(note.id, note.content, left, top);
   208→      }
   209→      isDragging.current = false;
   210→    };
   211→
   212→    document.addEventListener('touchmove', handleTouchMove, { passive: true });
   213→    document.addEventListener('touchend', handleTouchEnd);
   214→  }, [note.id, note.content, note.positionX, note.positionY, editing, onUpdate]);
   215→
   216→  const handleSave = () => {
   217→    onUpdate(note.id, editText, note.positionX, note.positionY);
   218→    setEditing(false);
   219→  };
   220→
   221→  return (
   222→    <div
   223→      ref={noteRef}
   224→      className="absolute z-50 select-none"
   225→      style={{ left: `${note.positionX}%`, top: `${note.positionY}%` }}
   226→    >
   227→      <div className="bg-yellow-100 border-2 border-yellow-300 rounded-lg shadow-lg p-2 min-w-[130px] max-w-[180px] transition-shadow hover:shadow-xl">
   228→        <div className="flex items-center justify-between mb-1">
   229→          <GripVertical className="h-3 w-3 text-yellow-400 cursor-grab active:cursor-grabbing" />
   230→          <div className="flex items-center gap-1">
   231→            <button
   232→              onClick={(e) => { e.stopPropagation(); if (editing) { handleSave(); } else { startEditing(); } }}
   233→              className="text-yellow-400 hover:text-[var(--theme-primary)]"
   234→            >
   235→              <PenLine className="h-2.5 w-2.5" />
   236→            </button>
   237→            <button
   238→              onClick={(e) => { e.stopPropagation(); onDelete(note.id); }}
   239→              className="text-yellow-400 hover:text-red-500"
   240→            >
   241→              <X className="h-2.5 w-2.5" />
   242→            </button>
   243→          </div>
   244→        </div>
   245→        {note.bibleRef && (
   246→          <p className="text-[8px] text-[var(--theme-primary)] font-bold mb-0.5">📖 {note.bibleRef}</p>
   247→        )}
   248→        {editing ? (
   249→          <textarea
   250→            value={editText}
   251→            onChange={(e) => setEditText(e.target.value)}
   252→            onBlur={handleSave}
   253→            onKeyDown={(e) => { if (e.key === 'Escape') setEditing(false); if (e.key === 'Enter' && e.metaKey) handleSave(); }}
   254→            className="w-full text-[10px] bg-transparent border border-yellow-200 rounded p-1 outline-none resize-none h-16 focus:border-[var(--theme-primary)]"
   255→            autoFocus
   256→            onClick={(e) => e.stopPropagation()}
   257→            onMouseDown={(e) => e.stopPropagation()}
   258→          />
   259→        ) : (
   260→          <p
   261→            onClick={(e) => { e.stopPropagation(); startEditing(); }}
   262→            className="text-[10px] text-gray-700 cursor-text leading-tight"
   263→          >
   264→            {note.content}
   265→          </p>
   266→        )}
   267→      </div>
   268→    </div>
   269→  );
   270→}

   804→              {noteContext.bibleRef ? `Note pour : ${noteContext.bibleRef}` : noteContext.bookId ? 'Note pour ce livre' : 'Note personnelle'}
   805→            </DialogDescription>
   806→          </DialogHeader>
   807→          {noteContext.bibleRef === undefined && !noteContext.bookId && (
   808→            <Select value={noteContext.bibleRef ? 'bible' : noteContext.bookId ? 'book' : 'general'} onValueChange={(v) => {
   809→              if (v === 'general') setNoteContext({});
   810→              else if (v === 'bible') setNoteContext({ bibleRef: 'Bible' });
   811→              else if (v === 'book' && books.length > 0) setNoteContext({ bookId: books[0].id });
   812→            }}>
   813→              <SelectTrigger className="h-9 text-xs"><SelectValue placeholder="Type de note" /></SelectTrigger>
   814→              <SelectContent>
   815→                <SelectItem value="general">📝 Note générale</SelectItem>
   816→                <SelectItem value="bible">📖 Note Bible</SelectItem>
   817→                {books.length > 0 && <SelectItem value="book">📚 Note livre</SelectItem>}
   818→              </SelectContent>
   819→            </Select>
   820→          )}
   821→          {(noteContext.bibleRef || noteContext.bibleRef === '') && noteContext.bookId === undefined && (
   822→            <Select value={noteContext.bibleRef || ''} onValueChange={(v) => setNoteContext({ bibleRef: v })}>
   823→              <SelectTrigger className="h-9 text-xs"><SelectValue placeholder="Référence biblique" /></SelectTrigger>
   824→              <SelectContent>
   825→                <div className="max-h-48 overflow-y-auto">
   826→                  {BIBLE_BOOKS.map((b) => (
   827→                    <SelectItem key={b.name} value={b.name} className="text-xs">{b.name}</SelectItem>
   828→                  ))}
   829→                </div>
   830→              </SelectContent>
   831→            </Select>
   832→          )}
   833→          {noteContext.bookId && books.length > 0 && !noteContext.bibleRef && (
   834→            <Select value={noteContext.bookId} onValueChange={(v) => setNoteContext({ bookId: v })}>
   835→              <SelectTrigger className="h-9 text-xs"><SelectValue placeholder="Choisir un livre" /></SelectTrigger>
   836→              <SelectContent>
   837→                {books.map((b) => (
   838→                  <SelectItem key={b.id} value={b.id} className="text-xs">{b.title}</SelectItem>
   839→                ))}
   840→              </SelectContent>
   841→            </Select>
   842→          )}
   843→          <Textarea
   844→            value={noteText}
   845→            onChange={(e) => setNoteText(e.target.value)}
   846→            placeholder="Écrivez votre note ici..."
   847→            className="min-h-[100px] text-sm"
   848→            autoFocus
   849→          />
   850→          <Button
   851→            onClick={() => {
   852→              if (!noteText.trim()) return;
   853→              // Random position to avoid overlap
   854→              const x = 10 + Math.random() * 60;
   855→              const y = 10 + Math.random() * 60;
   856→              addNote.mutate({ ...noteContext, content: noteText, positionX: x, positionY: y });
   857→            }}
   858→            disabled={!noteText.trim()}
   859→            className="w-full bg-[var(--theme-primary)] hover:bg-[var(--theme-primary-hover)] text-white text-xs"
   860→          >
   861→            <Plus className="h-3 w-3 mr-1" /> Ajouter la note
   862→          </Button>
   863→        </DialogContent>
   864→      </Dialog>
   865→
   866→      {/* PDF Viewer Dialog */}
   867→      <PdfViewerDialog
   868→        open={pdfViewerOpen}
   869→        onOpenChange={setPdfViewerOpen}
   870→        pdfUrl={pdfViewerUrl}
   871→        title={pdfViewerTitle}
   872→      />
   873→    </div>
   874→  );
   875→}
