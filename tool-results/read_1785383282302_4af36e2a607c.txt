   111→   111→// Floating Note Component
   112→   112→function FloatingNote({ note, onUpdate, onDelete, onDragStart }: {
   113→   113→  note: ReadingNote;
   114→   114→  onUpdate: (id: string, content: string) => void;
   115→   115→  onDelete: (id: string) => void;
   116→   116→  onDragStart: (id: string, e: React.MouseEvent) => void;
   117→   117→}) {
   118→   118→  const [editing, setEditing] = useState(false);
   119→   119→  const [text, setText] = useState(note.content);
   120→   120→
   121→   121→  const handleSave = () => {
   122→   122→    onUpdate(note.id, text);
   123→   123→    setEditing(false);
   124→   124→  };
   125→   125→
   126→   126→  return (
   127→   127→    <div
   128→   128→      className="absolute z-50 cursor-move select-none"
   129→   129→      style={{ left: `${note.positionX}%`, top: `${note.positionY}%` }}
   130→   130→      onMouseDown={(e) => onDragStart(note.id, e)}
   131→   131→    >
   132→   132→      <div className="bg-yellow-100 border border-yellow-300 rounded-lg shadow-lg p-2 min-w-[140px] max-w-[200px]">
   133→   133→        <div className="flex items-center justify-between mb-1">
   134→   134→          <GripVertical className="h-3 w-3 text-yellow-400" />
   135→   135→          <button onClick={() => onDelete(note.id)} className="text-yellow-400 hover:text-red-400">
   136→   136→            <X className="h-3 w-3" />
   137→   137→          </button>
   138→   138→        </div>
   139→   139→        {editing ? (
   140→   140→          <textarea value={text} onChange={(e) => setText(e.target.value)} onBlur={handleSave}
   141→   141→            onKeyDown={(e) => { if (e.key === 'Escape') setEditing(false); }}
   142→   142→            className="w-full text-[10px] bg-transparent border-none outline-none resize-none h-12"
   143→   143→            autoFocus />
   144→   144→        ) : (
   145→   145→          <p onClick={() => setEditing(true)} className="text-[10px] text-gray-700 cursor-text leading-tight">{note.content}</p>
   146→   146→        )}
   147→   147→        {note.bibleRef && <p className="text-[8px] text-yellow-600 mt-1 font-medium">📖 {note.bibleRef}</p>}
   148→   148→      </div>
   149→   149→    </div>
   150→   150→  );
   151→   151→}
